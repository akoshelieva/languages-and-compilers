struct Point
{
    i32 mut x
    fn shift := (i32 dx) -> i32
    {
        x := x + dx
        exit x
    }
}
struct Circle
{
    Point mut center
    i32 radius
    fn move := (i32 dx) -> i32
    {
        exit center.shift(dx)
    }
}
Point p{3}
Circle mut c{p, 2}
i32 a{c.move(4)}
i32 b{c.center.shift(1)}
i32 resultValue{a + b + c.center.x}
exit resultValue
